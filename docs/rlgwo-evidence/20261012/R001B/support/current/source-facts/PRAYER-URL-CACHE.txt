function readPrayerCache(key=cacheKey()){
  try{
    const j=JSON.parse(localStorage.getItem(key));
    if(recordObject(j)){ const admitted=admitPrayerRecord(j.data,j.date); if(admitted.ok) return admitted; }
  }catch(e){}
  return null;
}
function loadCache(dateStr,key=cacheKey()){
  const cached=readPrayerCache(key); return cached && cached.date===dateStr ? cached.data : null;
}
function saveCache(dateStr,data,key=cacheKey()){
  const admitted=admitPrayerRecord(data,dateStr); if(!admitted.ok) return;
  try{ localStorage.setItem(key,JSON.stringify({date:dateStr,data:admitted.data})); }catch(e){}
}
